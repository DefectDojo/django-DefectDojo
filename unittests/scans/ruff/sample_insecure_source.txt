"""Deliberately insecure sample used to generate a Ruff SARIF fixture."""
import hashlib
import pickle

import subprocess

import requests

def connect(password="placeholder-not-a-real-secret"):

def weak_digest(data):
    return hashlib.md5(data).hexdigest()


def pick():

def load(blob):
    return pickle.loads(blob)


def run(user_input):
    return subprocess.run(user_input, shell=True, check=False)

    return requests.get(url, verify=False, timeout=10)
